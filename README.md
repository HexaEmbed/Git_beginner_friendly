# Git_beginner_friendly project structure

# my-project-name/
# │
# ├── .git/                # Git metadata (auto-created when you run `git init`)
# ├── .gitignore           # Files/folders Git should ignore
# ├── README.md            # Project overview & usage instructions
# ├── LICENSE              # License information (optional)
# │
# ├── src/                 # Source code
# │   ├── main.c
# │   └── utils.c
# │
# ├── include/             # Header files
# │   └── utils.h
# │
# ├── tests/               # Unit tests
# │   └── test_utils.c
# │
# ├── docs/                # Documentation
# │   └── design.md
# │
# └── build/               # Compiled output (ignored in Git)
